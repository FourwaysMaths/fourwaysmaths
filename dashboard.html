async function generateGeminiQuiz() { const topicInput = document.getElementById("quiz-topic-input").value || "High School Mathematics"; const difficulty = document.getElementById("quiz-difficulty").value; const statusDiv = document.getElementById("quiz-status"); const containerDiv = document.getElementById("quiz-container"); statusDiv.innerHTML = `

Generating quiz with Gemini AI...

`; containerDiv.innerHTML = ""; try { const { data, error } = await supabaseClient.functions.invoke(GEMINI_FUNCTION_NAME, { body: { topic: topicInput, difficulty: difficulty, grade: "Grade 9" // Passed explicitly for your function prompt } }); if (error) throw error; statusDiv.innerHTML = ""; // Updated to access data.quiz instead of data.questions const rawQuizList = data?.quiz || [ { question: `Sample question on ${topicInput}: Simplify 2x + 3x`, options: ["5x", "6x", "5x²", "x"], answerIndex: 0, explanation: "Combine like terms: 2x + 3x = (2+3)x = 5x." } ]; // Map answerIndex to string answer for compatibility const quizList = rawQuizList.map(q => ({ ...q, answer: q.answer || (q.options && q.answerIndex !== undefined ? q.options[q.answerIndex] : "") })); containerDiv.innerHTML = quizList.map((q, idx) => `

Q${idx+1}: ${q.question}

${q.options.map(opt => ` `).join('')}
`).join(''); if (window.MathJax) MathJax.typesetPromise([containerDiv]).catch(() => {}); } catch (err) { console.error("Quiz Generation Error:", err); statusDiv.innerHTML = `

Failed to generate quiz. Try again.

`; } }
